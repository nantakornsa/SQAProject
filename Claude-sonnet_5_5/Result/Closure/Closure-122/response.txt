```java
package com.google.javascript.jscomp.parsing;

import com.google.javascript.jscomp.SourceFile;
import com.google.javascript.jscomp.parsing.Config.LanguageMode;
import com.google.javascript.rhino.Node;
import com.google.javascript.rhino.testing.TestErrorReporter;

import junit.framework.TestCase;

public class SuspiciousBlockCommentRegressionTest extends TestCase {

  private static final String SUSPICIOUS_COMMENT_WARNING =
      "Non-JSDoc comment has annotations. Did you mean to start it with '/**'?";

  private Node parse(String string, String... warnings) {
    TestErrorReporter testErrorReporter = new TestErrorReporter(null, warnings);
    Node script = null;
    try {
      script = ParserRunner.parse(
          SourceFile.fromCode("input", string),
          string,
          ParserRunner.createConfig(true, LanguageMode.ECMASCRIPT3, false, null),
          testErrorReporter).ast;
    } catch (Exception e) {
      throw new RuntimeException(e);
    }

    // verifying that all warnings were seen
    testErrorReporter.assertHasEncounteredAllWarnings();
    return script;
  }

  public void testSuspiciousBlockCommentWarningNoSpaceAfterStar() {
    parse("/* \n *@type {number} */ var x = 3;", SUSPICIOUS_COMMENT_WARNING);
  }

  public void testSuspiciousBlockCommentWarningTabIndent() {
    parse("/* \n\t* @type {number} */ var x = 3;", SUSPICIOUS_COMMENT_WARNING);
  }

  public void testSuspiciousBlockCommentWarningSpaceBeforeStar() {
    parse("/*\n   * @type {number} */ var x = 3;", SUSPICIOUS_COMMENT_WARNING);
  }

  public void testSuspiciousBlockCommentWarningNoSpaceAfterOpening() {
    parse("/*@type {number} */ var x = 3;", SUSPICIOUS_COMMENT_WARNING);
  }
}
```

I could not check this against the project's sources, so it may need small adjustments to compile and run.

- **API signatures:** The `ParserRunner.createConfig(true, LanguageMode.ECMASCRIPT3, false, null)` and `ParserRunner.parse(...)` signatures are written from memory of the mid-2013 code. They may differ slightly in this revision.
- **Warning text:** The `SUSPICIOUS_COMMENT_WARNING` string is also from memory and needs to match the message in `IRFactory`.
- **Other tests:** I did not look at the existing `ParserTest` helper. I wrote a similar `parse` helper using `TestErrorReporter` instead.

The four tests use comments with an annotation after `/*` or after a newline plus optional whitespace and `*`, for example `/* \n *@type {number} */`. The fixed regex `(/|(\n[ \t]*))\*[ \t]*@[a-zA-Z]` accepts all of them and raises the warning. The buggy check only looked for the literal strings `/* @` and `\n * @`, so it misses them and `assertHasEncounteredAllWarnings()` fails.
