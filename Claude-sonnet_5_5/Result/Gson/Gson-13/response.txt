package com.google.gson.stream;

import java.io.StringReader;
import junit.framework.TestCase;

public final class JsonReaderNegativeZeroRegressionTest extends TestCase {

  public void testNegativeZeroNextString() throws Exception {
    JsonReader reader = new JsonReader(new StringReader("[-0]"));
    reader.setLenient(false);
    reader.beginArray();
    assertEquals(JsonToken.NUMBER, reader.peek());
    assertEquals("-0", reader.nextString());
    reader.endArray();
  }

  public void testNegativeZeroInArrayWithOtherValues() throws Exception {
    JsonReader reader = new JsonReader(new StringReader("[-0, 0, -1]"));
    reader.beginArray();
    assertEquals(JsonToken.NUMBER, reader.peek());
    assertEquals("-0", reader.nextString());
    assertEquals(JsonToken.NUMBER, reader.peek());
    assertEquals("0", reader.nextString());
    assertEquals(JsonToken.NUMBER, reader.peek());
    assertEquals("-1", reader.nextString());
    reader.endArray();
  }
}
