package com.google.gson;

import java.text.ParsePosition;
import java.util.Date;

import com.google.gson.internal.bind.util.ISO8601Utils;

import junit.framework.TestCase;

/**
 * Regression test for ISO8601 dates whose timezone offset has no minutes (e.g. "+01").
 */
public class ISO8601TimezoneWithoutMinutesRegressionTest extends TestCase {

  public void testParseTimezoneOffsetWithoutMinutes() throws Exception {
    Date parsed = ISO8601Utils.parse("1970-01-01T01:00:00+01", new ParsePosition(0));
    assertEquals(0L, parsed.getTime());
  }

  public void testParseTimezoneOffsetWithMinutesStillWorks() throws Exception {
    assertEquals(0L,
        ISO8601Utils.parse("1970-01-01T01:00:00+01:00", new ParsePosition(0)).getTime());
    assertEquals(0L,
        ISO8601Utils.parse("1970-01-01T01:00:00+0100", new ParsePosition(0)).getTime());
  }

  public void testGsonDeserializesIso8601WithShortTimezoneOffset() throws Exception {
    Gson gson = new Gson();
    Date date = gson.fromJson("\"1970-01-01T01:00:00+01\"", Date.class);
    assertEquals(0L, date.getTime());
  }
}
