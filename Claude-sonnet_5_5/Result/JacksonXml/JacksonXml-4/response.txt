package com.fasterxml.jackson.dataformat.xml.misc;

import java.io.IOException;

import com.fasterxml.jackson.databind.ObjectWriter;
import com.fasterxml.jackson.dataformat.xml.XmlMapper;
import com.fasterxml.jackson.dataformat.xml.XmlTestBase;

public class RootNameNullDynamicTest extends XmlTestBase
{
    private final XmlMapper _xmlMapper = new XmlMapper();

    // [dataformat-xml#213]: explicitly configured root name must be used for null value too
    public void testDynamicRootNameWithNullValue() throws IOException
    {
        ObjectWriter w = _xmlMapper.writer().withRootName("rudy");
        String xml = w.writeValueAsString(null);
        assertEquals("<rudy/>", xml);
    }
}
