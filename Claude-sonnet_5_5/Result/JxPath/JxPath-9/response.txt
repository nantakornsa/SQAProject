package org.apache.commons.jxpath.ri.compiler;

import junit.framework.TestCase;

import org.apache.commons.jxpath.JXPathContext;

/**
 * Regression test for JXPATH-95: comparisons involving NaN must never
 * evaluate to true, including "=" and "!=".
 */
public class CoreOperationNanRegressionTest extends TestCase {

    private JXPathContext context;

    protected void setUp() throws Exception {
        super.setUp();
        context = JXPathContext.newContext(null);
        context.getVariables().declareVariable("nan", new Double(Double.NaN));
    }

    private void assertFalseValue(String xpath) {
        Object result = context.getValue(xpath);
        assertEquals("Evaluating <" + xpath + ">", Boolean.FALSE, result);
    }

    public void testNanEqualsNan() {
        assertFalseValue("$nan = $nan");
    }

    public void testNanNotEqualsNan() {
        assertFalseValue("$nan != $nan");
    }

    public void testNanComparedWithNumbers() {
        assertFalseValue("$nan = 0");
        assertFalseValue("$nan != 0");
        assertFalseValue("$nan = 1");
        assertFalseValue("$nan != 1");
    }

    public void testNanRelationalOperators() {
        assertFalseValue("$nan > $nan");
        assertFalseValue("$nan < $nan");
        assertFalseValue("$nan >= $nan");
        assertFalseValue("$nan <= $nan");
        assertFalseValue("$nan >= $nan and $nan <= $nan");
        assertFalseValue("$nan > 0");
        assertFalseValue("$nan < 0");
        assertFalseValue("$nan >= 0 and $nan <= 0");
    }
}
