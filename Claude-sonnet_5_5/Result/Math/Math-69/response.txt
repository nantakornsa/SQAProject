package org.apache.commons.math.stat.correlation;

import junit.framework.TestCase;

/**
 * Regression test for MATH-371: p-values computed by PearsonsCorrelation
 * vanished prematurely because of loss of precision in 1 - cumulativeProbability(t).
 */
public class PearsonsCorrelationPValueRegressionTest extends TestCase {

    public void testPValueNearZero() throws Exception {
        /*
         * Create a dataset that has r -> 1, p -> 0 as dimension increases.
         * Prior to the fix for MATH-371, p vanished for dimension >= 14.
         * Post fix, p-values diminish smoothly, vanishing at dimension = 127.
         */
        int dimension = 120;
        double[][] data = new double[dimension][2];
        for (int i = 0; i < dimension; i++) {
            data[i][0] = i;
            data[i][1] = i + 1 / ((double) i + 1);
        }
        PearsonsCorrelation corrInstance = new PearsonsCorrelation(data);
        double p = corrInstance.getCorrelationPValues().getEntry(0, 1);
        assertTrue("p-value should be strictly positive", p > 0);
        assertTrue("p-value should be very small", p < 1E-10);
    }
}
