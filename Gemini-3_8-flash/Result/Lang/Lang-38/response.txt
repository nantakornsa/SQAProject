package org.apache.commons.lang3.time;

import java.util.Calendar;
import java.util.GregorianCalendar;
import java.util.TimeZone;
import junit.framework.TestCase;

public class FastDateFormatLang538Test extends TestCase {

    public void testLang538() {
        final String expectedDateTime = "2009-10-16T16:42:16.000Z";

        // Constructing calendar in GMT-8 with fields set but without calling getTime() or computeFields()
        Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("GMT-8"));
        cal.clear();
        cal.set(2009, Calendar.OCTOBER, 16, 8, 42, 16);

        // Formatting with forced timezone GMT
        FastDateFormat format = FastDateFormat.getInstance("yyyy-MM-dd'T'HH:mm:ss.SSS'Z'", TimeZone.getTimeZone("GMT"));
        assertEquals("dateTime", expectedDateTime, format.format(cal));
    }
}
