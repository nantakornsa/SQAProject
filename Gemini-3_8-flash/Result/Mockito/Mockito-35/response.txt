package org.mockitousage.bugs;

import org.junit.Test;
import org.mockito.Mockito;

import static org.mockito.Matchers.eq;
import static org.mockito.Matchers.isA;
import static org.mockito.Matchers.same;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.verify;

public class MatchersNPEOnPrimitivesTest {

    interface IMethods {
        void intArgumentMethod(int value);
    }

    @Test
    public void shouldNotThrowNPEWhenPrimitiveIntPassedToIsA() {
        IMethods mock = mock(IMethods.class);
        mock.intArgumentMethod(100);

        verify(mock).intArgumentMethod(isA(Integer.class));
    }

    @Test
    public void shouldNotThrowNPEWhenPrimitiveIntPassedToEq() {
        IMethods mock = mock(IMethods.class);
        mock.intArgumentMethod(100);

        verify(mock).intArgumentMethod(eq(100));
    }

    @Test
    public void shouldNotThrowNPEWhenPrimitiveIntPassedToSame() {
        IMethods mock = mock(IMethods.class);
        mock.intArgumentMethod(100);

        verify(mock).intArgumentMethod(same(100));
    }
}
